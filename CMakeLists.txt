cmake_minimum_required(VERSION 3.25)
project(fnv_xbox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(REXGLUE_PLATFORM_SWITCH)
    set(FNV_HORIZON_SDK "" CACHE PATH "ReXGlue Horizon fork source tree")
    if(NOT EXISTS "${FNV_HORIZON_SDK}/cmake/rexglue_switch.cmake")
        message(FATAL_ERROR "Set FNV_HORIZON_SDK to the documented ReXGlue Horizon fork, with third-party sources fetched.")
    endif()
    add_subdirectory("${FNV_HORIZON_SDK}" "${CMAKE_BINARY_DIR}/horizon-sdk")
elseif(WIN32)
    find_package(rexglue 0.10.0 EXACT CONFIG REQUIRED)
else()
    message(FATAL_ERROR "Supported targets: Windows reference or Horizon using cmake/switch-devkitA64.cmake.")
endif()
set(FNV_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/default" CACHE PATH "Private codegen directory")
if(NOT EXISTS "${FNV_GENERATED_DIR}/sources.cmake")
    message(FATAL_ERROR "Run tools/codegen.ps1 using your own Xbox 360 files first.")
endif()
include("${FNV_GENERATED_DIR}/sources.cmake")
if(REXGLUE_PLATFORM_SWITCH)
    set(_trace_default OFF)
else()
    set(_trace_default ON)
endif()
option(FNV_HOST_BOOT_TRACE "Compile private observers and opt-in startup barrier" ${_trace_default})
if(REXGLUE_PLATFORM_SWITCH AND FNV_HOST_BOOT_TRACE)
    message(FATAL_ERROR "Windows boot observers must be disabled for Horizon.")
endif()
if(FNV_HOST_BOOT_TRACE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_boot_trace.py"
        --generated-dir "${FNV_GENERATED_DIR}" COMMAND_ERROR_IS_FATAL ANY)
    foreach(partition 105 129 12 228 172 25)
        set(original "${FNV_GENERATED_DIR}/fnv_xbox_recomp.${partition}.cpp")
        set(observed "${CMAKE_CURRENT_SOURCE_DIR}/build/boot-trace/fnv_xbox_recomp.${partition}.cpp")
        list(REMOVE_ITEM GENERATED_SOURCES "${original}")
        list(APPEND GENERATED_SOURCES "${observed}")
    endforeach()
endif()
add_library(fnv_xbox_recomp OBJECT ${GENERATED_SOURCES})
target_include_directories(fnv_xbox_recomp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${FNV_GENERATED_DIR}")
target_link_libraries(fnv_xbox_recomp PRIVATE rex::runtime)
rexglue_apply_target_settings(fnv_xbox_recomp)
target_precompile_headers(fnv_xbox_recomp PRIVATE "${FNV_GENERATED_DIR}/fnv_xbox_pch.h")
if(REXGLUE_PLATFORM_SWITCH)
    set(FNV_SWITCH_GUEST_OPTIMIZATION "2" CACHE STRING "GCC optimization level for private guest code")
    if(NOT FNV_SWITCH_GUEST_OPTIMIZATION MATCHES "^[0123]$")
        message(FATAL_ERROR "FNV_SWITCH_GUEST_OPTIMIZATION must be 0, 1, 2 or 3")
    endif()
    target_compile_options(fnv_xbox_recomp PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/src/switch_codegen_compat.h" -ffp-contract=off -fno-strict-aliasing)
    target_compile_options(fnv_xbox_recomp PRIVATE "-O${FNV_SWITCH_GUEST_OPTIMIZATION}")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(fnv_xbox_recomp PRIVATE -fasync-exceptions -gline-tables-only)
elseif(MSVC)
    target_compile_options(fnv_xbox_recomp PRIVATE /EHa)
endif()
if(REXGLUE_PLATFORM_SWITCH)
    add_executable(fnv_xbox src/main.cpp src/switch_startup_gate.cpp src/switch_call_profile.cpp src/switch_render_mode.cpp)
    set_target_properties(fnv_xbox PROPERTIES OUTPUT_NAME "fnv-nx")
else()
    add_executable(fnv_xbox WIN32 src/main.cpp src/host_crash.cpp src/frame_watch.cpp src/test_pad.cpp)
endif()
target_include_directories(fnv_xbox PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${FNV_GENERATED_DIR}")
target_link_libraries(fnv_xbox PRIVATE fnv_xbox_recomp rex::runtime)
rexglue_configure_target(fnv_xbox GPU_PLUGINS xenos)
if(REXGLUE_PLATFORM_SWITCH)
    rexglue_switch_add_nro(fnv_xbox NAME "FNV-nx" AUTHOR "CypherNoodle" VERSION "${PROJECT_VERSION}" ICON "${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg")
    # The SDK packages by CMake target name, independently of OUTPUT_NAME.
    add_custom_command(TARGET fnv_xbox POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE_DIR:fnv_xbox>/fnv_xbox.nro"
            "$<TARGET_FILE_DIR:fnv_xbox>/fnv-nx.nro"
        VERBATIM)
endif()
