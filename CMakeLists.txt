cmake_minimum_required(VERSION 3.25)
project(fnv_xbox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
    message(FATAL_ERROR "This initial target is the Windows reference. Horizon integration is pending.")
endif()
find_package(rexglue 0.10.0 EXACT CONFIG REQUIRED)
set(FNV_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/default" CACHE PATH "Private codegen directory")
if(NOT EXISTS "${FNV_GENERATED_DIR}/sources.cmake")
    message(FATAL_ERROR "Run tools/codegen.ps1 using your own Xbox 360 files first.")
endif()
include("${FNV_GENERATED_DIR}/sources.cmake")
option(FNV_HOST_BOOT_TRACE "Compile private observers and opt-in startup barrier" ON)
if(FNV_HOST_BOOT_TRACE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_boot_trace.py"
        --generated-dir "${FNV_GENERATED_DIR}" COMMAND_ERROR_IS_FATAL ANY)
    foreach(partition 105 129 12 228 172 25)
        set(original "${FNV_GENERATED_DIR}/fnv_xbox_recomp.${partition}.cpp")
        set(observed "${CMAKE_CURRENT_SOURCE_DIR}/build/boot-trace/fnv_xbox_recomp.${partition}.cpp")
        list(REMOVE_ITEM GENERATED_SOURCES "${original}")
        list(APPEND GENERATED_SOURCES "${observed}")
    endforeach()
endif()
add_library(fnv_xbox_recomp OBJECT ${GENERATED_SOURCES})
target_include_directories(fnv_xbox_recomp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${FNV_GENERATED_DIR}")
target_link_libraries(fnv_xbox_recomp PRIVATE rex::runtime)
rexglue_apply_target_settings(fnv_xbox_recomp)
target_precompile_headers(fnv_xbox_recomp PRIVATE "${FNV_GENERATED_DIR}/fnv_xbox_pch.h")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(fnv_xbox_recomp PRIVATE -fasync-exceptions -gline-tables-only)
elseif(MSVC)
    target_compile_options(fnv_xbox_recomp PRIVATE /EHa)
endif()
add_executable(fnv_xbox WIN32 src/main.cpp src/host_crash.cpp src/frame_watch.cpp src/test_pad.cpp)
target_include_directories(fnv_xbox PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${FNV_GENERATED_DIR}")
target_link_libraries(fnv_xbox PRIVATE fnv_xbox_recomp rex::runtime)
rexglue_configure_target(fnv_xbox GPU_PLUGINS xenos)
