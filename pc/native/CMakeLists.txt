if(FNV_MINIMP3_ROOT)
  if(NOT EXISTS "${FNV_MINIMP3_ROOT}/minimp3_ex.h")
    message(FATAL_ERROR "FNV_MINIMP3_ROOT must contain pinned minimp3 headers")
  endif()
  foreach(FNV_MINIMP3_FILE_HASH IN ITEMS
      "minimp3.h=57e437c5c1f0e8b243885d3929c8973b5e6c778451e0100ab4251d19915cb3ad"
      "minimp3_ex.h=8437f3fc1d4d8ab2269a1624f5380a08df1967a048f8887789bae6e25db7db79"
      "LICENSE=6a1ee543e5282cd9061881edf462e6fdab181f328da71fc2c9a6950a80e94d01")
    string(REPLACE "=" ";" FNV_MINIMP3_PAIR "${FNV_MINIMP3_FILE_HASH}")
    list(GET FNV_MINIMP3_PAIR 0 FNV_MINIMP3_FILE)
    list(GET FNV_MINIMP3_PAIR 1 FNV_MINIMP3_EXPECTED)
    file(SHA256 "${FNV_MINIMP3_ROOT}/${FNV_MINIMP3_FILE}" FNV_MINIMP3_ACTUAL)
    if(NOT FNV_MINIMP3_ACTUAL STREQUAL FNV_MINIMP3_EXPECTED)
      message(FATAL_ERROR "minimp3 dependency differs from pinned revision: ${FNV_MINIMP3_FILE}")
    endif()
  endforeach()
  add_library(fnv_mp3_stream STATIC mp3_stream.c)
  target_include_directories(fnv_mp3_stream SYSTEM PRIVATE ${FNV_MINIMP3_ROOT})

    # generated target too: generic ARM64 defaults read the wrong TP register.

      -mtp=soft -ftls-model=local-exec -fPIE)
